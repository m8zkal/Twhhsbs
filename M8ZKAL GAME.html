// ================================================================ // PvP (simple-peer + Firebase) – исправленная версия // ================================================================ let pvpProcessingSignal = false; function createPvpRoom() { if (!db) { alert('Firebase недоступен'); return; } const code = generateRoomCode(); pvpRoomCode = code; pvpIsCreator = true; pvpSelectionButtons.style.display = 'none'; pvpCodeEntry.style.display = 'block'; pvpCharacterBlock.style.display = 'none'; pvpPasteBtn.style.display = 'none'; pvpClearBtn.style.display = 'none'; pvpCopyBtn.style.display = 'block'; pvpRoomInput.value = code; pvpRoomInput.disabled = true; pvpCancelBtn.style.display = 'block'; pvpReadyBtn.style.display = 'block'; pvpReadyBtn.disabled = false; pvpReadyBtn.classList.remove('ready'); pvpRoomRef = db.ref('rooms/' + code); pvpRoomRef.set({ status: 'waiting', creatorReady: false, joinerReady: false, offer: null, answer: null, ice: [] }); pvpPeer = new SimplePeer({ initiator: true, trickle: true, config: { iceServers: [{ urls: 'stun:stun.l.google.com:19302' }] } }); setupPvpPeerEvents(); // Слушаем изменения в комнате pvpRoomRef.on('value', function(snapshot) { const data = snapshot.val(); if (!data) return; if (data.status === 'ready') { if (!pvpConnected) startHockeyDuel(); } // Обрабатываем сигналы ТОЛЬКО если соединение ещё не установлено if (!pvpConnected && !pvpProcessingSignal && pvpPeer && !pvpPeer.destroyed) { if (pvpIsCreator && data.answer) { pvpProcessingSignal = true; try { pvpPeer.signal(JSON.parse(data.answer)); // Сразу удаляем answer, чтобы не обрабатывать повторно pvpRoomRef.child('answer').set(null); } catch(e) { console.warn('signal(answer) error:', e); } pvpProcessingSignal = false; } if (!pvpIsCreator && data.offer) { pvpProcessingSignal = true; try { pvpPeer.signal(JSON.parse(data.offer)); } catch(e) { console.warn('signal(offer) error:', e); } pvpProcessingSignal = false; } if (data.ice && data.ice.length > 0) { data.ice.forEach(function(candidate) { if (!pvpConnected && pvpPeer && !pvpPeer.destroyed) { try { pvpPeer.signal(candidate); } catch(e) {} } }); // Очищаем ice после обработки pvpRoomRef.child('ice').set([]); } } }); } function joinPvpRoom(code) { if (!db) { alert('Firebase недоступен'); return; } if (pvpPeer) return; pvpRoomCode = code; pvpIsCreator = false; pvpSelectionButtons.style.display = 'none'; pvpCodeEntry.style.display = 'block'; pvpCharacterBlock.style.display = 'none'; pvpPasteBtn.style.display = 'block'; pvpClearBtn.style.display = 'block'; pvpCopyBtn.style.display = 'block'; pvpRoomInput.value = code; pvpRoomInput.disabled = true; pvpCancelBtn.style.display = 'block'; pvpReadyBtn.style.display = 'block'; pvpReadyBtn.disabled = false; pvpRoomRef = db.ref('rooms/' + code); pvpRoomRef.once('value', function(snapshot) { const data = snapshot.val(); if (!data || data.status === 'waiting') { pvpPeer = new SimplePeer({ initiator: false, trickle: true, config: { iceServers: [{ urls: 'stun:stun.l.google.com:19302' }] } }); setupPvpPeerEvents(); // Слушаем изменения комнаты pvpRoomRef.on('value', function(snap) { const d = snap.val(); if (!d) return; if (d.status === 'ready') { if (!pvpConnected) startHockeyDuel(); } if (!pvpConnected && !pvpProcessingSignal && pvpPeer && !pvpPeer.destroyed) { if (!pvpIsCreator && d.offer) { pvpProcessingSignal = true; try { pvpPeer.signal(JSON.parse(d.offer)); } catch(e) { console.warn('signal(offer) error:', e); } pvpProcessingSignal = false; } if (pvpIsCreator && d.answer) { pvpProcessingSignal = true; try { pvpPeer.signal(JSON.parse(d.answer)); pvpRoomRef.child('answer').set(null); } catch(e) { console.warn('signal(answer) error:', e); } pvpProcessingSignal = false; } if (d.ice && d.ice.length > 0) { d.ice.forEach(function(c) { if (!pvpConnected && pvpPeer && !pvpPeer.destroyed) { try { pvpPeer.signal(c); } catch(e) {} } }); pvpRoomRef.child('ice').set([]); } } }); pvpRoomRef.update({ status: 'joined' }); } else { alert('Комната не найдена или уже занята.'); cancelPvp(); } }); } function setupPvpPeerEvents() { pvpPeer.on('signal', function(data) { if (pvpPeer.destroyed) return; // Отправляем сигналы только если соединение ещё не установлено if (!pvpConnected) { if (data.type === 'offer') { if (pvpRoomRef) pvpRoomRef.child('offer').set(JSON.stringify(data)); } else if (data.type === 'answer') { if (pvpRoomRef) pvpRoomRef.child('answer').set(JSON.stringify(data)); } else if (data.candidate) { if (pvpRoomRef) pvpRoomRef.child('ice').push(data); } } }); pvpPeer.on('connect', function() { console.log('PvP соединение установлено!'); pvpConnected = true; // Очищаем все сигнальные данные, чтобы не мешали if (pvpRoomRef) { pvpRoomRef.child('offer').set(null); pvpRoomRef.child('answer').set(null); pvpRoomRef.child('ice').set([]); } if (!pvpIsCreator) { pvpRoomRef.update({ joinerReady: true }); pvpRoomRef.once('value', function(snap) { const d = snap.val(); if (d && d.creatorReady && d.joinerReady) pvpRoomRef.update({ status: 'ready' }); }); } else { pvpRoomRef.update({ creatorReady: true }); pvpRoomRef.once('value', function(snap) { const d = snap.val(); if (d && d.creatorReady && d.joinerReady) pvpRoomRef.update({ status: 'ready' }); }); } }); pvpPeer.on('data', function(data) { try { const parsed = JSON.parse(data); handlePvpData(parsed); } catch(e) { console.warn('Ошибка парсинга данных:', e); } }); pvpPeer.on('error', function(err) { console.error('PvP ошибка:', err); alert('Ошибка соединения: ' + err.message); cancelPvp(); goToMenu(); }); pvpPeer.on('close', function() { pvpConnected = false; console.log('PvP соединение закрыто'); cancelPvp(); }); }
